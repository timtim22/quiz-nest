import { z } from 'zod';
import type { UserRole } from '../constants/roles.js';

/** bcrypt only looks at the first 72 bytes of a password. Urdu letters take 2+ bytes each. */
export const PASSWORD_MAX_BYTES = 72;
export const PASSWORD_MIN_LENGTH = 8;

/** Length of a string in UTF-8 bytes, which is what bcrypt counts. */
export function utf8ByteLength(value: string): number {
  let bytes = 0;
  for (const char of value) {
    const code = char.codePointAt(0) ?? 0;
    bytes += code < 0x80 ? 1 : code < 0x800 ? 2 : code < 0x10000 ? 3 : 4;
  }
  return bytes;
}

const email = z
  .string()
  .trim()
  .toLowerCase()
  .pipe(z.email('Enter a valid email address.').max(254, 'Email is too long.'));

/** Rules for a new password. Login uses a looser check so old passwords still work. */
const newPassword = z
  .string()
  .min(PASSWORD_MIN_LENGTH, `Use at least ${PASSWORD_MIN_LENGTH} characters.`)
  .refine((value) => utf8ByteLength(value) <= PASSWORD_MAX_BYTES, {
    message: 'Password is too long.',
  })
  .refine((value) => /\p{L}/u.test(value) && /\d/.test(value), {
    message: 'Use at least one letter and one number.',
  });

export const registerSchema = z.object({
  name: z.string().trim().min(2, 'Enter your name.').max(100, 'Name is too long.'),
  email,
  password: newPassword,
  schoolName: z
    .string()
    .trim()
    .min(2, 'Enter your school or academy name.')
    .max(120, 'School name is too long.'),
});

export const loginSchema = z.object({
  email,
  password: z
    .string()
    .min(1, 'Enter your password.')
    .refine((value) => utf8ByteLength(value) <= PASSWORD_MAX_BYTES, {
      message: 'Password is too long.',
    }),
});

export const forgotPasswordSchema = z.object({ email });

export const resetPasswordSchema = z.object({
  token: z.string().min(1, 'This reset link is invalid.').max(200),
  password: newPassword,
});

export type RegisterInput = z.infer<typeof registerSchema>;
export type LoginInput = z.infer<typeof loginSchema>;
export type ForgotPasswordInput = z.infer<typeof forgotPasswordSchema>;
export type ResetPasswordInput = z.infer<typeof resetPasswordSchema>;

/** The logged-in user, as the web app sees them. */
export interface AuthUser {
  id: string;
  name: string;
  email: string;
  role: UserRole;
  organizationId: string;
  organizationName: string;
}

/** Returned by register, login and refresh. The refresh token travels separately in a cookie. */
export interface AuthSession {
  accessToken: string;
  user: AuthUser;

import { Injectable, type OnModuleDestroy } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { PrismaPg } from '@prisma/adapter-pg';
import type { Env } from '../config/env.js';
import { PrismaClient } from '../generated/prisma/client.js';

/**
 * The database client. Inject it into any service:
 *
 *   constructor(private readonly prisma: PrismaService) {}
 *
 * Connections open lazily on the first query, so the API can start (and report
 * the problem on /api/health) even when the database is down.
 */
@Injectable()
export class PrismaService extends PrismaClient implements OnModuleDestroy {
  constructor(config: ConfigService<Env, true>) {
    super({
      adapter: new PrismaPg({ connectionString: config.get('DATABASE_URL', { infer: true }) }),
    });
  }

  async onModuleDestroy() {
    await this.$disconnect();
  }
}

import { Body, Controller, Get, HttpCode, Post, Req, Res, UseGuards } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { Throttle, ThrottlerGuard } from '@nestjs/throttler';
import {
  type AuthSession,
  type AuthUser,
  type ForgotPasswordInput,
  forgotPasswordSchema,
  type LoginInput,
  loginSchema,
  type RegisterInput,
  registerSchema,
  type ResetPasswordInput,
  resetPasswordSchema,
} from '@quiz-nest/shared';
import type { CookieOptions, Request, Response } from 'express';
import { API_PREFIX } from '../../app.setup.js';
import { ZodValidationPipe } from '../../common/pipes/zod-validation.pipe.js';
import type { Env } from '../../config/env.js';
import { REFRESH_COOKIE, REFRESH_TOKEN_TTL_MS } from './auth.constants.js';
import { AuthService } from './auth.service.js';
import type { RequestUser } from './auth.types.js';
import { CurrentUser, Public } from './decorators.js';
import { PasswordResetService } from './password-reset.service.js';

const MINUTE = 60_000;

/**
 * Teacher/admin accounts. Login returns a short-lived access token in the body and
 * sets a long-lived refresh token in an httpOnly cookie that only /api/auth receives.
 * The cookie is SameSite=Strict, so other websites can't make the browser send it.
 */
@Controller('auth')
@UseGuards(ThrottlerGuard)
export class AuthController {
  constructor(
    private readonly auth: AuthService,
    private readonly passwordReset: PasswordResetService,
    private readonly config: ConfigService<Env, true>,
  ) {}

  @Public()
  @Throttle({ default: { limit: 5, ttl: 10 * MINUTE } })
  @Post('register')
  async register(
    @Body(new ZodValidationPipe(registerSchema)) body: RegisterInput,
    @Res({ passthrough: true }) res: Response,
  ): Promise<AuthSession> {
    const { session, refreshToken } = await this.auth.register(body);
    this.setRefreshCookie(res, refreshToken);
    return session;
  }

  @Public()
  @Throttle({ default: { limit: 5, ttl: MINUTE } })
  @Post('login')
  @HttpCode(200)
  async login(
    @Body(new ZodValidationPipe(loginSchema)) body: LoginInput,
    @Res({ passthrough: true }) res: Response,
  ): Promise<AuthSession> {
    const { session, refreshToken } = await this.auth.login(body);
    this.setRefreshCookie(res, refreshToken);
    return session;
  }

  /**
   * The web app calls this on page load and whenever the access token expires.
   * Returns null when there's no cookie at all: being logged out isn't an error.
   */
  @Public()
  @Throttle({ default: { limit: 30, ttl: MINUTE } })
  @Post('refresh')
  @HttpCode(200)
  async refresh(
    @Req() req: Request,
    @Res({ passthrough: true }) res: Response,
  ): Promise<AuthSession | null> {
    const cookie: string | undefined = req.cookies?.[REFRESH_COOKIE];
    if (!cookie) return null;

    try {
      const { session, refreshToken } = await this.auth.refresh(cookie);
      this.setRefreshCookie(res, refreshToken);
      return session;
    } catch (error) {
      this.clearRefreshCookie(res);
      throw error;
    }
  }

  @Public()
  @Post('logout')
  @HttpCode(200)
  async logout(@Req() req: Request, @Res({ passthrough: true }) res: Response): Promise<null> {
    await this.auth.logout(req.cookies?.[REFRESH_COOKIE]);
    this.clearRefreshCookie(res);
    return null;
  }

  @Public()
  @Throttle({ default: { limit: 3, ttl: MINUTE } })
  @Post('forgot-password')
  @HttpCode(200)
  async forgotPassword(
    @Body(new ZodValidationPipe(forgotPasswordSchema)) body: ForgotPasswordInput,
  ): Promise<null> {
    await this.passwordReset.requestReset(body.email);
    return null;
  }

  @Public()
  @Throttle({ default: { limit: 5, ttl: MINUTE } })
  @Post('reset-password')
  @HttpCode(200)
  async resetPassword(
    @Body(new ZodValidationPipe(resetPasswordSchema)) body: ResetPasswordInput,
    @Res({ passthrough: true }) res: Response,
  ): Promise<null> {
    await this.passwordReset.resetPassword(body.token, body.password);
    this.clearRefreshCookie(res);
    return null;
  }

  @Get('me')
  me(@CurrentUser() user: RequestUser): Promise<AuthUser> {
    return this.auth.me(user.id);
  }

  private cookieOptions(): CookieOptions {
    return {
      httpOnly: true,
      secure: this.config.get('NODE_ENV', { infer: true }) === 'production',
      sameSite: 'strict',
      path: `/${API_PREFIX}/auth`,
    };
  }

  private setRefreshCookie(res: Response, token: string) {
    res.cookie(REFRESH_COOKIE, token, { ...this.cookieOptions(), maxAge: REFRESH_TOKEN_TTL_MS });
  }

  private clearRefreshCookie(res: Response) {
    res.clearCookie(REFRESH_COOKIE, this.cookieOptions());
  }
}

// A 401 from these means "wrong password" or "logged out", not "token expired",
// so they never trigger the automatic refresh-and-retry.
const noRetry = { retryOnUnauthorized: false };

export const register = (input: RegisterInput) =>
  apiClient.post<AuthSession>('/auth/register', input, noRetry);

export const login = (input: LoginInput) =>
  apiClient.post<AuthSession>('/auth/login', input, noRetry);

/** Uses the httpOnly refresh cookie, which the browser sends automatically. Null = logged out. */
export const refresh = () =>
  apiClient.post<AuthSession | null>('/auth/refresh', undefined, noRetry);

export const logout = () => apiClient.post<null>('/auth/logout', undefined, noRetry);

export const forgotPassword = (input: ForgotPasswordInput) =>
  apiClient.post<null>('/auth/forgot-password', input, noRetry);

export const resetPassword = (input: ResetPasswordInput) =>
  apiClient.post<null>('/auth/reset-password', input, noRetry);

export const getMe = () => apiClient.get<AuthUser>('/auth/me');

import type { AuthSession } from '@quiz-nest/shared';
import { useQueryClient } from '@tanstack/react-query';
import { type ReactNode, useEffect, useMemo, useState } from 'react';
import { setAccessToken, setAccessTokenRefresher } from '@/lib/api-client';
import { logout } from './api';
import { type AuthContextValue, AuthContext, type AuthState } from './auth-context';
import { restoreSession } from './session';

/** Knows who is logged in. Wraps the whole app (see App.tsx). */
export function AuthProvider({ children }: { children: ReactNode }) {
  const queryClient = useQueryClient();
  const [state, setState] = useState<AuthState>({ status: 'loading' });

  useEffect(() => {
    let active = true;
    const apply = (session: AuthSession | null) => {
      if (!active) return;
      setState(session ? { status: 'authenticated', user: session.user } : { status: 'anonymous' });
    };

    // Access tokens last 15 minutes. When one expires, the API client calls this and retries.
    setAccessTokenRefresher(async () => {
      const session = await restoreSession().catch(() => null);
      apply(session);
      return session?.accessToken ?? null;
    });

    // On page load, the refresh cookie tells us whether the teacher is still logged in.
    restoreSession().then(apply, () => apply(null));

    return () => {
      active = false;
      setAccessTokenRefresher(null);
    };
  }, []);

  const value = useMemo<AuthContextValue>(
    () => ({
      ...state,
      signIn(session) {
        setAccessToken(session.accessToken);
        setState({ status: 'authenticated', user: session.user });
      },
      async signOut() {
        try {
          await logout();
        } finally {
          setAccessToken(null);
          queryClient.clear();
          setState({ status: 'anonymous' });
        }
      },
    }),
    [state, queryClient],
  );

  return <AuthContext value={value}>{children}</AuthContext>;
}

import type { AuthSession } from '@quiz-nest/shared';
import { ApiError, setAccessToken } from '@/lib/api-client';
import { refresh } from './api';

let inFlight: Promise<AuthSession | null> | null = null;

/**
 * Gets a fresh access token using the refresh cookie. Returns null when the user
 * isn't logged in. Calls made while one is already running share it, so the cookie
 * is only swapped once (React StrictMode and parallel requests both do this).
 */
export function restoreSession(): Promise<AuthSession | null> {
  inFlight ??= refresh()
    .then((session) => {
      setAccessToken(session?.accessToken ?? null);
      return session;
    })
    .catch((error: unknown) => {
      setAccessToken(null);
      if (error instanceof ApiError && (error.status === 401 || error.status === 403)) return null;
      throw error;
    })
    .finally(() => {
      inFlight = null;
    });

  return inFlight;
}

/** Only allow redirects within this app, e.g. `/teacher/exams`, never `//evil.com`. */
export function safeNextPath(next: string | null, fallback = '/teacher/dashboard'): string {
  if (!next || !next.startsWith('/') || next.startsWith('//') || next.startsWith('/\\')) {
    return fallback;
  }
  return next;
}

import { type ApiResponse, ErrorCode } from '@quiz-nest/shared';

const BASE_URL = import.meta.env.VITE_API_URL ?? '/api';

/** Thrown for every failed request. Check `code` (an ErrorCode) to decide what to show. */
export class ApiError extends Error {
  readonly code: ErrorCode;
  /** HTTP status, or 0 when the request never reached the server. */
  readonly status: number;
  readonly details?: unknown;

  constructor(code: ErrorCode, message: string, status: number, details?: unknown) {
    super(message);
    this.name = 'ApiError';
    this.code = code;
    this.status = status;
    this.details = details;
  }
}

type QueryParams = Record<string, string | number | boolean | undefined | null>;

export interface RequestOptions extends Omit<RequestInit, 'body' | 'method'> {
  /** Sent as JSON. */
  body?: unknown;
  query?: QueryParams;
  /**
   * On a 401, get a new access token and try once more (default true).
   * Auth calls set this to false: a wrong password must not trigger a refresh.
   */
  retryOnUnauthorized?: boolean;
}

// The access token lives only in memory, never in localStorage, so other scripts
// on the page can't steal it. After a page reload the auth feature gets a new one
// using the refresh cookie.
let accessToken: string | null = null;
let accessTokenRefresher: (() => Promise<string | null>) | null = null;

/** Set by the auth feature after login, refresh and logout. */
export function setAccessToken(token: string | null) {
  accessToken = token;
}

/**
 * Set by the auth feature. Called when a request gets a 401: it should return a new
 * access token, or null if the user is logged out. The request is then retried once.
 */
export function setAccessTokenRefresher(refresher: (() => Promise<string | null>) | null) {
  accessTokenRefresher = refresher;
}

/**
 * The only place the web app talks to the API. Use it from feature `api.ts` files:
 *
 *   export const getSubjects = () => apiClient.get<Subject[]>('/subjects');
 *
 * It returns the `data` of { success: true, data } and throws ApiError otherwise.
 */
export const apiClient = {
  get: <T>(path: string, options?: RequestOptions) => request<T>('GET', path, options),
  post: <T>(path: string, body?: unknown, options?: RequestOptions) =>
    request<T>('POST', path, { ...options, body }),
  put: <T>(path: string, body?: unknown, options?: RequestOptions) =>
    request<T>('PUT', path, { ...options, body }),
  patch: <T>(path: string, body?: unknown, options?: RequestOptions) =>
    request<T>('PATCH', path, { ...options, body }),
  delete: <T>(path: string, options?: RequestOptions) => request<T>('DELETE', path, options),
};

async function request<T>(method: string, path: string, options: RequestOptions = {}): Promise<T> {
  const { retryOnUnauthorized = true, ...sendOptions } = options;

  try {
    return await send<T>(method, path, sendOptions);
  } catch (error) {
    const expired = error instanceof ApiError && error.status === 401;
    if (expired && retryOnUnauthorized && accessTokenRefresher && (await accessTokenRefresher())) {
      return send<T>(method, path, sendOptions);
    }
    throw error;
  }
}

async function send<T>(method: string, path: string, options: RequestOptions): Promise<T> {
  const { body, query, headers: extraHeaders, ...init } = options;

  const headers = new Headers(extraHeaders);
  headers.set('Accept', 'application/json');
  if (body !== undefined) headers.set('Content-Type', 'application/json');
  if (accessToken) headers.set('Authorization', `Bearer ${accessToken}`);

  let response: Response;
  try {
    response = await fetch(buildUrl(path, query), {
      ...init,
      method,
      headers,
      body: body === undefined ? undefined : JSON.stringify(body),
      credentials: 'include',
    });
  } catch (error) {
    throw new ApiError(
      ErrorCode.CONNECTION_UNAVAILABLE,
      'Could not reach the server. Check your internet connection.',
      0,
      error,
    );
  }

  const payload = await readJson<ApiResponse<T>>(response);

  if (payload?.success === true && response.ok) return payload.data;
  if (payload?.success === false) {
    const { code, message, details } = payload.error;
    throw new ApiError(code, message, response.status, details);
  }
  throw new ApiError(
    response.status >= 500 ? ErrorCode.SERVICE_UNAVAILABLE : ErrorCode.BAD_REQUEST,
    `Unexpected response from the server (${response.status}).`,
    response.status,
  );
}

function buildUrl(path: string, query?: QueryParams): string {
  const url = `${BASE_URL}${path.startsWith('/') ? path : `/${path}`}`;
  if (!query) return url;

  const params = new URLSearchParams();
  for (const [key, value] of Object.entries(query)) {
    if (value !== undefined && value !== null) params.set(key, String(value));
  }
  const search = params.toString();
  return search ? `${url}?${search}` : url;
}

async function readJson<T>(response: Response): Promise<T | undefined> {
  try {
    return (await response.json()) as T;
  } catch {
    return undefined;
  }
}
